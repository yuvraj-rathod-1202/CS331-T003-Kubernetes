% SLIDE 2 — Problem Statement & Objective
\begin{frame}{Problem Statement \& Objective}
\footnotesize

\textbf{Problem:} Kubernetes heals container crashes but is \alert{blind to network-layer failures}.

\begin{itemize}
  \item CNI plugin crashes $\to$ pods stuck in \texttt{ContainerCreating}
  \item CoreDNS degradation $\to$ silent DNS resolution failures
  \item Overlay tunnel drops $\to$ cross-node traffic silently fails
  \item Pods show \texttt{Running (1/1 Ready)} while network is dead
\end{itemize}

\vspace{6pt}

\begin{block}{Objective}
Build a \textbf{custom Kubernetes operator} that continuously monitors and auto-remediates three critical network subsystems:
\begin{enumerate}
  \item \textbf{CNI plugin health} (Calico IPAM exhaustion \& agent failures)
  \item \textbf{CoreDNS availability} (scale-to-zero, corrupted upstreams, latency spikes)
  \item \textbf{Pod-to-pod connectivity} (overlay tunnel drops \& cross-node routing)
\end{enumerate}
\end{block}

\end{frame}

% SLIDE 5 — Proposed Solution: Custom Kubernetes Operator
\begin{frame}{Extension: Custom Kubernetes Operator}
\footnotesize

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Operator Architecture \& Control Plane:}
\begin{itemize}
  \item \textbf{Custom Go Operator:} Built with Kubebuilder; extends Kubernetes self-healing directly into the network data plane.
  \item \textbf{Cluster-Scoped CRD:} Single declarative \texttt{NetworkRemediation} (\texttt{v1alpha1}) resource governing all monitoring and remediation.
  \item \textbf{Modular Subsystem Toggles:} Independent \texttt{enabled} flags for CNI, CoreDNS, and Pod Connectivity modules with zero coupling.
  \item \textbf{Three-Phase Pipeline:} Standardized \emph{Check $\to$ Evaluate $\to$ Remediate} reconciliation executed every 30\,s.
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Core Capabilities \& Safety Guardrails:}
\begin{itemize}
  \item \textbf{$O(N)$ Ring Probing:} Adapted Pingmesh topology ($2N$ probes vs.\ $O(N^2)$ full mesh) with intra-node canary validation.
  \item \textbf{Configurable CRD Specs:} Users can define module specifications and detection thresholds directly in the CRD.
  \item \textbf{Safety Guardrails:} 30\,s cooldown windows, consecutive failure thresholds ($N \ge 3$), and bounded restart escalation.
\end{itemize}
\end{column}
\end{columns}

\end{frame}

% SLIDE 8 — Non-Functional Testing Parameters
\begin{frame}{Non-Functional Testing Parameters}
\footnotesize

\begin{columns}[T]
\begin{column}{0.48\textwidth}
\textbf{Performance \& Reliability:}
\begin{itemize}
  \item \textbf{Remediation time:} 30-60\,s across all modules
  \item \textbf{Reconcile interval:} 30\,s (configurable)
  \item \textbf{Cooldown windows:} 30\,s per remediation type to prevent restart thrashing
  \item \textbf{Consecutive failure threshold:} 3 (avoids reacting to transient issues)
\end{itemize}
\end{column}

\begin{column}{0.48\textwidth}
\textbf{Safety \& Resource Efficiency:}
\begin{itemize}
  \item \textbf{Independent toggles:} each remediation type gated separately
  \item \textbf{Single operator pod:} manages all three modules
  \item \textbf{Minimal footprint:} no additional DaemonSet or sidecar required
\end{itemize}
\end{column}
\end{columns}

  \vspace{6pt}

  \begin{itemize}\footnotesize

  \end{itemize}

  \vspace{6pt}
  \item \textbf{Failure of Reconcile Loop Cooldown:}
  \begin{itemize}\footnotesize
    \item Duplicate restarts before network convergence.
    \item Resolved via 30\,s cooldowns and sequential pipeline execution.
  \end{itemize}
